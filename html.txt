function shortest(from,to){
  const dist={},prev={},left=new Set(Object.keys(NODES)); for(const n in NODES) dist[n]=Infinity; dist[from]=0;
  while(left.size){ let u=null; for(const n of left) if(u===null||dist[n]<dist[u]) u=n;
    left.delete(u); if(u===to||dist[u]===Infinity) break;
    for(const e of ADJ[u]||[]){ const d=dist[u]+len(e.pts); if(d<dist[e.to]){dist[e.to]=d;prev[e.to]={from:u,e}} } }
  const path=[]; let n=to; while(n!==from){ const p=prev[n]; if(!p) return []; path.unshift({from:p.from,to:n,pts:p.e.pts}); n=p.from; }
  return path;
}

function turnOf(a,b){ // a: 들어오는 마지막 구간, b: 나가는 첫 구간
  const v1=[a[1][0]-a[0][0],a[1][1]-a[0][1]], v2=[b[1][0]-b[0][0],b[1][1]-b[0][1]];
  const cr=v1[0]*v2[1]-v1[1]*v2[0], dot=v1[0]*v2[0]+v1[1]*v2[1];
  if(Math.abs(cr)<1e-6&&dot>0) return 'S'; return cr>0?'R':'L';
}

function seatLayout(office){

  isl.forEach((I,ii)=>{
    const cols=I.C?['한 줄']:['왼쪽','오른쪽'], w=cols.length*CW;

    const top=PANEL_H+10+(I.head?CH+10:0);
    if(I.head) cells.push({isl:ii+1,col:'맨 앞',n:0,x:I.head==='C'?x+(w-CW)/2:x+CW,y:PANEL_H+10});

/* 보여줄 영역(box)을 그림 칸의 가로세로 비율에 맞춰 viewBox로 */
function fitView(S,box,minW,minH,shrink){
  let [x0,y0,x1,y1]=box, w=x1-x0, h=y1-y0;
  if(w<minW){x0-=(minW-w)/2;w=minW}
  if(h<(minH||0)){y0-=(minH-h)/2;h=minH}
  if(shrink){ // 가로로 긴 도면: 칸 높이를 그림에 맞춰 줄이고 남는 자리는 다른 그림에 줌
    const fig=S.parentNode, fr=fig.getBoundingClientRect(), need=fr.width*h/w;
    if(need<fr.height){ fig.style.flex='0 0 auto'; fig.style.height=Math.round(need)+'px'; } }
  const r=S.getBoundingClientRect(), A=(r.width||4)/(r.height||3);
  if(w/h<A){const nw=h*A;x0-=(nw-w)/2;w=nw}else{const nh=w/A;y0-=(nh-h)/2;h=nh}
  S.setAttribute('viewBox',[x0,y0,w,h].map(v=>v.toFixed(1)).join(' '));
  return w/(r.width||1); // 화면 1px이 몇 단위인지 (표시 크기 맞춤용)
}

/* ── 서버(Apps Script) 부르기: GET 요청 → { ok, data } ── */
function api(query,ok,fail){
  if(!API_URL||API_URL.includes('배포ID')){ fail(new Error('index.html 위쪽의 API_URL에 Apps Script 웹앱 주소를 넣어 주세요.')); return; }
  const ctl=new AbortController(), tm=setTimeout(()=>ctl.abort(),25000);
  fetch(API_URL+(API_URL.includes('?')?'&':'?')+query,{cache:'no-store',signal:ctl.signal})
    .then(r=>{ if(!r.ok) throw new Error('서버 응답 '+r.status); return r.json(); })
    .then(j=>{ clearTimeout(tm); j&&j.ok?ok(j.data):fail(new Error(j&&j.error||'알 수 없는 오류')); })
    .catch(e=>{ clearTimeout(tm); fail(e.name==='AbortError'?new Error('응답이 너무 늦어요'):e); });
}

/* 화면 한글 키보드: 입력된 글자를 자모로 풀고, 누른 자모를 더한 뒤 다시 조합 */
const HG={
  CHO:'ㄱㄲㄴㄷㄸㄹㅁㅂㅃㅅㅆㅇㅈㅉㅊㅋㅌㅍㅎ',
  JUNG:'ㅏㅐㅑㅒㅓㅔㅕㅖㅗㅘㅙㅚㅛㅜㅝㅞㅟㅠㅡㅢㅣ',
  JONG:' ㄱㄲㄳㄴㄵㄶㄷㄹㄺㄻㄼㄽㄾㄿㅀㅁㅂㅄㅅㅆㅇㅈㅊㅋㅌㅍㅎ',
  V2:{'ㅗㅏ':'ㅘ','ㅗㅐ':'ㅙ','ㅗㅣ':'ㅚ','ㅜㅓ':'ㅝ','ㅜㅔ':'ㅞ','ㅜㅣ':'ㅟ','ㅡㅣ':'ㅢ'},
  C2:{'ㄱㅅ':'ㄳ','ㄴㅈ':'ㄵ','ㄴㅎ':'ㄶ','ㄹㄱ':'ㄺ','ㄹㅁ':'ㄻ','ㄹㅂ':'ㄼ','ㄹㅅ':'ㄽ','ㄹㅌ':'ㄾ','ㄹㅍ':'ㄿ','ㄹㅎ':'ㅀ','ㅂㅅ':'ㅄ'}
};
HG.V2r={}; for(const k in HG.V2) HG.V2r[HG.V2[k]]=k;
HG.C2r={}; for(const k in HG.C2) HG.C2r[HG.C2[k]]=k;
const isV=j=>HG.JUNG.includes(j);
function hgDecompose(str){
  const out=[];
  for(const ch of str){ const c=ch.charCodeAt(0)-0xAC00;
    if(c>=0&&c<11172){ const cho=HG.CHO[Math.floor(c/588)], jung=HG.JUNG[Math.floor(c%588/28)], jong=HG.JONG[c%28];
      out.push(cho); out.push(...(HG.V2r[jung]||jung).split(''));
      if(jong!==' ') out.push(...(HG.C2r[jong]||jong).split('')); }
    else if(HG.V2r[ch]) out.push(...HG.V2r[ch].split(''));
    else if(HG.C2r[ch]) out.push(...HG.C2r[ch].split(''));
    else out.push(ch);
  }
  return out;
}
function hgCompose(jamo){
  let out='', s=null;
  const flush=()=>{ if(!s) return;
    if(s.cho&&s.jung) out+=String.fromCharCode(0xAC00+HG.CHO.indexOf(s.cho)*588+HG.JUNG.indexOf(s.jung)*28+(s.jong?HG.JONG.indexOf(s.jong):0));
    else out+=(s.cho||'')+(s.jung||'');
    s=null; };
  for(const j of jamo){
    if(isV(j)){
      if(s&&s.cho&&!s.jung) s.jung=j;
      else if(s&&s.jung&&!s.jong&&HG.V2[s.jung+j]) s.jung=HG.V2[s.jung+j];
      else if(s&&s.jong){ const pair=HG.C2r[s.jong]; let move;
        if(pair){ s.jong=pair[0]; move=pair[1]; } else { move=s.jong; s.jong=null; }
        flush(); s={cho:move,jung:j};
      } else { flush(); s={jung:j}; }
    } else if(HG.CHO.includes(j)){
      if(s&&s.cho&&s.jung&&!s.jong&&HG.JONG.includes(j)) s.jong=j;
      else if(s&&s.jong&&HG.C2[s.jong+j]) s.jong=HG.C2[s.jong+j];
      else { flush(); s={cho:j}; }
    } else { flush(); out+=j; }
  }
  flush(); return out;
}
